Sub xuanxing()
    Dim neiji As Long, waiji As Long, baojia As Long
    Dim NeijiFirstRow As Long, NeijiLastRow As Long, WaijiLastRow As Long, WaijiFirstRow As Long, BaojiaFirstRow As Long, BaojiaLastRow As Long
    Dim lengliang As Double, neijixilie As String
    Dim i As Long
    Dim neijishuliang As Long, neijizuidalengliang As Double, neijilengliang As Double, neijixinghao As String
    Dim ws As Worksheet
    Dim xitonglengliang As Double
    Dim waijixilie As String
    Dim waijixinghao As String
    Dim mergeStart As Long
    Dim currentValue As String
    Dim chaopeilv As Double
    Dim anzhuang As Long  ' 添加这个变量声明
    
    '基本参数赋值
    Set ws = ThisWorkbook.Sheets("空调报价表")
    neijixilie = [I1].Value
    neijizuidalengliang = [K1].Value
    waijixilie = [D2].Value
    chaopeilv = [F2].Value
    Dim waijishuliang As Long
    
    '获取需要编辑选型的区域
    With ActiveSheet.Columns(1)
        neiji = .Find("一、空调设备内机配置详表", LookAt:=xlWhole).Row
        waiji = .Find("二、空调设备外机配置详表", LookAt:=xlWhole).Row
        baojia = .Find("三、空调设备部分报价表", LookAt:=xlWhole).Row
        anzhuang = .Find("四、运输、安装人工费", LookAt:=xlWhole).Row
    End With
    NeijiFirstRow = neiji + 3
    NeijiLastRow = waiji - 1
    
    If ws.Range("C" & NeijiFirstRow + 1).Value = "" Then
        MsgBox "请至少输入2个房间的面积信息"
        Exit Sub
    End If
    
    ' ========== 修改1：删除行 → 改为清空内容（删除内机区域空白行） ==========
    For i = NeijiFirstRow To NeijiLastRow
        If ws.Range("C" & i).Value = "" Then
            ' 原代码：ws.Rows(i & ":" & NeijiLastRow).Delete
            ' 改为：仅清空内容，不删除行
            ws.Rows(i & ":" & NeijiLastRow).ClearContents
            Exit For
        End If
    Next i
    
    With ActiveSheet.Columns(1)
        neiji = .Find("一、空调设备内机配置详表", LookAt:=xlWhole).Row
        waiji = .Find("二、空调设备外机配置详表", LookAt:=xlWhole).Row
        baojia = .Find("三、空调设备部分报价表", LookAt:=xlWhole).Row
    End With
    NeijiFirstRow = neiji + 3
    NeijiLastRow = waiji - 1
    WaijiFirstRow = waiji + 2
    WaijiLastRow = baojia - 1
    
    ' 内机选型
    For i = NeijiFirstRow To NeijiLastRow
        ' 跳过空行
        If ws.Range("C" & i).Value <> "" Then
            neijishuliang = ws.Range("E" & i).Value
            neijilengliang = ws.Range("C" & i).Value * ws.Range("M" & i).Value / 1000 / neijishuliang
            ws.Range("D" & i).Value = GetNeijixinghao(neijixilie, neijilengliang)
        End If
    Next i
    
    '外机选型
    ws.Range("K" & NeijiFirstRow & ":K" & NeijiLastRow).UnMerge
    mergeStart = NeijiFirstRow
    currentValue = ws.Cells(mergeStart, "L").Value
    waijishuliang = 0
    xitonglengliang = 0
    
    ' 先计算外机数量，用于确定写入位置
    Dim tempCount As Long
    tempCount = 0
    For i = NeijiFirstRow + 1 To NeijiLastRow
        If ws.Cells(i, "L").Value <> currentValue Then
            tempCount = tempCount + 1
            currentValue = ws.Cells(i, "L").Value
        End If
    Next i
    tempCount = tempCount + 1 ' 最后一组
    
    ' 确保外机区域有足够的行，如果不够则添加
    Dim currentRows As Long
    currentRows = WaijiLastRow - WaijiFirstRow + 1
    If tempCount > currentRows Then
        ws.Rows(WaijiLastRow + 1 & ":" & WaijiLastRow + (tempCount - currentRows)).Insert
        WaijiLastRow = WaijiLastRow + (tempCount - currentRows)
    End If
    
    ' 重新获取外机区域范围
    With ActiveSheet.Columns(1)
        waiji = .Find("二、空调设备外机配置详表", LookAt:=xlWhole).Row
        baojia = .Find("三、空调设备部分报价表", LookAt:=xlWhole).Row
    End With
    WaijiFirstRow = waiji + 2
    WaijiLastRow = baojia - 1
    
    ' 外机选型主循环
    mergeStart = NeijiFirstRow
    currentValue = ws.Cells(mergeStart, "L").Value
    waijishuliang = 0
    xitonglengliang = 0
    
    For i = NeijiFirstRow + 1 To NeijiLastRow
        ' 跳过空行
        If ws.Range("C" & i).Value <> "" Then
            xitonglengliang = xitonglengliang + ws.Range("J" & (i - 1)).Value
        End If
        
        If ws.Cells(i, "L").Value <> currentValue Then
            waijishuliang = waijishuliang + 1
            If waijishuliang <= WaijiLastRow - WaijiFirstRow + 1 Then
                ws.Range("K" & mergeStart).Value = GetWaijixinghao(waijixilie, xitonglengliang, chaopeilv)
                ws.Range("K" & mergeStart & ":K" & (i - 1)).Merge
                ws.Range("C" & WaijiFirstRow + waijishuliang - 1).Formula = "=K" & mergeStart
                ws.Range("B" & WaijiFirstRow + waijishuliang - 1).Formula = "=A" & mergeStart
                ws.Range("G" & WaijiFirstRow + waijishuliang - 1).Formula = "=SUM(J" & mergeStart & ":J" & (i - 1) & ")/1000"
            End If
            mergeStart = i
            currentValue = ws.Cells(i, "L").Value
            xitonglengliang = 0
        End If
        
        If i = NeijiLastRow Then
            If ws.Range("C" & i).Value <> "" Then
                xitonglengliang = xitonglengliang + ws.Range("J" & i).Value
            End If
            ws.Range("K" & mergeStart & ":K" & NeijiLastRow).Merge
            waijishuliang = waijishuliang + 1
            If waijishuliang <= WaijiLastRow - WaijiFirstRow + 1 Then
                ws.Range("K" & mergeStart).Value = GetWaijixinghao(waijixilie, xitonglengliang, chaopeilv)
                ws.Range("G" & WaijiFirstRow + waijishuliang - 1).Formula = "=SUM(J" & mergeStart & ":J" & i & ")/1000"
                ws.Range("C" & WaijiFirstRow + waijishuliang - 1).Formula = "=K" & mergeStart
                ws.Range("B" & WaijiFirstRow + waijishuliang - 1).Formula = "=A" & mergeStart
            End If
        End If
    Next i
    
    ' ========== 修改2：删除行 → 改为清空内容（外机区域） ==========
    For i = WaijiFirstRow To WaijiLastRow
        If ws.Range("C" & i).Value = "" Then
            ' 原代码：ws.Rows(i & ":" & WaijiLastRow).Delete
            ' 改为：仅清空内容，不删除行
            ws.Rows(i & ":" & WaijiLastRow).ClearContents
            Exit For
        End If
    Next i
    
    '整理设备报价区域，删除多余行(反向循环)
    With ActiveSheet.Columns(1)
        baojia = .Find("三、空调设备部分报价表", LookAt:=xlWhole).Row
        anzhuang = .Find("四、运输、安装人工费", LookAt:=xlWhole).Row
    End With
    BaojiaFirstRow = baojia + 2
    BaojiaLastRow = anzhuang - 3
    
    ' ========== 修改3：删除行 → 改为清空内容（报价区域） ==========
    For i = BaojiaLastRow To BaojiaFirstRow Step -1
        If ws.Range("L" & i).Value = "" Then
            ' 原代码：ws.Rows(i).Delete
            ' 改为：仅清空内容，不删除行
            ws.Rows(i).ClearContents
        End If
    Next i
    
End Sub

' 以下两个函数保持不变
'输出内机的选型型号
Function GetNeijixinghao(neijixilie As String, neijilengliang As Double) As String
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    
    Set ws = ThisWorkbook.Sheets("设备参数表")
    lastRow = ws.Cells(ws.Rows.Count, "L").End(xlUp).Row
    
    For i = 5 To lastRow
        If ws.Cells(i, "L").Value = neijixilie And ws.Cells(i, "G").Value >= neijilengliang Then
            GetNeijixinghao = ws.Cells(i, "D").Value
            Exit Function
        End If
    Next i
    
    GetNeijixinghao = "错误" ' 如果没有找到符合条件的值，返回空字符串
End Function

'输出外机的选型型号
Function GetWaijixinghao(waijixilie As String, xitonglengliang As Double, chaopeilv As Double) As String
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    
    Set ws = ThisWorkbook.Sheets("设备参数表")
    lastRow = 210
    
    For i = 5 To lastRow
        If ws.Cells(i, "L").Value = waijixilie And ws.Cells(i, "G").Value * chaopeilv >= xitonglengliang / 1000 Then
            GetWaijixinghao = ws.Cells(i, "D").Value
            Exit Function
        End If
    Next i
    
    GetWaijixinghao = "该系统冷量超过所选外机最大冷量，请更换外机系列或重新划分系统"
End Function
